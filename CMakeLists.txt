cmake_minimum_required(VERSION 3.20)
project(KspMissionEphemeris LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ksp_ephemeris src/KspMission.Ephemeris/Ephemeris.cpp)
target_include_directories(ksp_ephemeris PUBLIC src/KspMission.Ephemeris)
add_executable(ksp_ephemeris_tests tests/KspMission.Ephemeris.Tests/Tests.cpp)
target_link_libraries(ksp_ephemeris_tests PRIVATE ksp_ephemeris)
enable_testing()
add_test(NAME ksp_ephemeris_tests COMMAND ksp_ephemeris_tests)
add_library(ksp_spacecraft src/KspMission.Spacecraft/Spacecraft.cpp)
target_include_directories(ksp_spacecraft PUBLIC src/KspMission.Spacecraft)
target_link_libraries(ksp_spacecraft PUBLIC ksp_ephemeris)
add_executable(ksp_spacecraft_tests tests/KspMission.Spacecraft.Tests/Tests.cpp)
target_link_libraries(ksp_spacecraft_tests PRIVATE ksp_spacecraft)
add_test(NAME ksp_spacecraft_tests COMMAND ksp_spacecraft_tests)
add_library(ksp_search src/KspMission.Search/Search.cpp)
target_include_directories(ksp_search PUBLIC src/KspMission.Search)
target_link_libraries(ksp_search PUBLIC ksp_ephemeris)
add_executable(ksp_search_tests tests/KspMission.Search.Tests/Tests.cpp)
target_link_libraries(ksp_search_tests PRIVATE ksp_search)
add_test(NAME ksp_search_tests COMMAND ksp_search_tests)
add_library(ksp_runtime_reader src/KspMission.RuntimeReader/RuntimeReader.cpp src/KspMission.RuntimeReader/Sha256.cpp)
target_include_directories(ksp_runtime_reader PUBLIC src/KspMission.RuntimeReader third_party)
target_link_libraries(ksp_runtime_reader PUBLIC ksp_spacecraft)
add_executable(ksp_runtime_reader_tests tests/KspMission.RuntimeReader.Tests/Tests.cpp)
target_link_libraries(ksp_runtime_reader_tests PRIVATE ksp_runtime_reader)
add_test(NAME ksp_runtime_reader_tests COMMAND ksp_runtime_reader_tests)
add_library(ksp_search_grid src/KspMission.Search/SearchGrid.cpp)
target_include_directories(ksp_search_grid PUBLIC src/KspMission.Search)
target_link_libraries(ksp_search_grid PUBLIC ksp_search)
add_executable(ksp_search_grid_tests tests/KspMission.SearchGrid.Tests/Tests.cpp)
target_link_libraries(ksp_search_grid_tests PRIVATE ksp_search_grid)
add_test(NAME ksp_search_grid_tests COMMAND ksp_search_grid_tests)
add_library(ksp_refine src/KspMission.Search/Refine.cpp)
target_include_directories(ksp_refine PUBLIC src/KspMission.Search)
target_link_libraries(ksp_refine PUBLIC ksp_spacecraft)
add_executable(ksp_refine_tests tests/KspMission.Refine.Tests/Tests.cpp)
target_link_libraries(ksp_refine_tests PRIVATE ksp_refine)
add_test(NAME ksp_refine_tests COMMAND ksp_refine_tests)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(GTKMM4 QUIET IMPORTED_TARGET gtkmm-4.0)
  pkg_check_modules(EPOXY QUIET IMPORTED_TARGET epoxy)
  if(TARGET PkgConfig::GTKMM4 AND TARGET PkgConfig::EPOXY)
    add_executable(ksp_desktop src/KspMission.Desktop/Main.cpp)
    target_link_libraries(ksp_desktop PRIVATE ksp_runtime_reader PkgConfig::GTKMM4 PkgConfig::EPOXY)
  endif()
endif()
add_library(ksp_worker_protocol src/KspMission.Worker/Worker.cpp)
target_include_directories(ksp_worker_protocol PUBLIC src/KspMission.Worker third_party)
target_link_libraries(ksp_worker_protocol PUBLIC ksp_search_grid ksp_refine ksp_runtime_reader)
add_executable(ksp_worker src/KspMission.Worker/Main.cpp)
target_link_libraries(ksp_worker PRIVATE ksp_worker_protocol)
add_executable(ksp_worker_tests tests/KspMission.Worker.Tests/Tests.cpp)
target_link_libraries(ksp_worker_tests PRIVATE ksp_worker_protocol)
add_test(NAME ksp_worker_tests COMMAND ksp_worker_tests)
add_library(ksp_mission_route src/KspMission.Search/MissionRoute.cpp)
target_include_directories(ksp_mission_route PUBLIC src/KspMission.Search)
target_link_libraries(ksp_mission_route PUBLIC ksp_search_grid)
add_executable(ksp_mission_route_tests tests/KspMission.MissionRoute.Tests/Tests.cpp)
target_link_libraries(ksp_mission_route_tests PRIVATE ksp_mission_route)
add_test(NAME ksp_mission_route_tests COMMAND ksp_mission_route_tests)
